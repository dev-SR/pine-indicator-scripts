// COMPUTE PROFILE
// ──────────────────────────────────────────────────────────────────────────────
// FIX ②: inner loop rewritten with integer row-index math instead of scanning
// all rows for every bar.  For each bar we compute its bottom row index and top
// row index in O(1), then only iterate the rows it actually overlaps.
// Worst case is still O(rows) per bar, but the constant factor is much smaller
// and the typical case (bar spans a few rows) is far faster.
// ══════════════════════════════════════════════════════════════════════════════
compute_profile(float[] p_h, float[] p_l, float[] p_v, float[] p_b,
                 int rows, string vol_type, float va_pct) =>
    sz   = array.size(p_h)
    float hi  = array.max(p_h)
    float lo  = array.min(p_l)
    float rng = hi - lo
    float[] tops = array.new_float(rows, 0.0)
    float[] bots = array.new_float(rows, 0.0)
    float[] vols = array.new_float(rows, 0.0)
    if rng <= 0 or sz == 0
        [tops, bots, vols, hi, hi, lo]
    else
        row_h = rng / rows
        inv_row_h = 1.0 / row_h          // FIX ②a: precompute reciprocal once
        for i = 0 to rows - 1
            array.set(tops, i, lo + (i + 1) * row_h)
            array.set(bots, i, lo +  i      * row_h)
        for b = 0 to sz - 1
            bh   = array.get(p_h, b)
            bl   = array.get(p_l, b)
            bv   = array.get(p_v, b)
            bbuy = array.get(p_b, b)
            br   = bh - bl
            use_v = vol_type == "Delta" ? math.abs(bbuy - (bv - bbuy)) : bv

            // FIX ②b: compute the row index range this bar spans directly,
            // avoiding a full O(rows) scan for every bar.
            i_bot = int(math.max((bl - lo) * inv_row_h, 0))
            i_top = int(math.min((bh - lo) * inv_row_h, rows - 1))

            if br <= 0
                // zero-range bar: dump all volume into its single row
                mid_i = int((bl + bh) * 0.5 - lo) * inv_row_h
                safe_i = int(math.max(math.min(mid_i, rows - 1), 0))
                array.set(vols, safe_i, array.get(vols, safe_i) + use_v)
            else
                inv_br = 1.0 / br         // FIX ②c: hoist division out of inner loop
                for i = i_bot to i_top
                    rt = array.get(tops, i)
                    rb = array.get(bots, i)
                    ov = math.max(0.0, math.min(bh, rt) - math.max(bl, rb))
                    if ov > 0
                        array.set(vols, i, array.get(vols, i) + use_v * ov * inv_br)

        poc_i = array.indexof(vols, array.max(vols))
        poc_p = (array.get(tops, poc_i) + array.get(bots, poc_i)) / 2
        tot   = array.sum(vols)
        tgt   = tot * va_pct / 100.0
        va_v  = array.get(vols, poc_i)
        va_top = poc_i
        va_bot = poc_i
        while va_v < tgt
            can_up = va_top < rows - 1
            can_dn = va_bot > 0
            up_v   = can_up ? array.get(vols, va_top + 1) : -1.0
            dn_v   = can_dn ? array.get(vols, va_bot - 1) : -1.0
            if not can_up and not can_dn
                break
            else if up_v >= dn_v
                va_top += 1
                va_v   += up_v
            else
                va_bot -= 1
                va_v   += dn_v
        vah_p = array.get(tops, va_top)
        val_p = array.get(bots, va_bot)
        [tops, bots, vols, poc_p, vah_p, val_p]

// ══════════════════════════════════════════════════════════════════════════════
// DRAW PROFILE BOXES
// ══════════════════════════════════════════════════════════════════════════════
draw_boxes(float[] tops, float[] bots, float[] vols,
             float poc_p, float vah_p, float val_p,
             int t1, int t2,
             color bar_css, color va_css, color poc_css,
             box[] bxs) =>
    n        = array.size(vols)
    max_v    = array.max(vols)
    bar_span = int((t2 - t1) * profile_width_pct / 100)
    for i = 0 to n - 1
        rv  = array.get(vols, i)
        if rv <= 0
            continue
        rt  = array.get(tops, i)
        rb  = array.get(bots, i)
        mid = (rt + rb) / 2
        in_va  = mid >= val_p and mid <= vah_p
        is_poc = math.abs(mid - poc_p) < (rt - rb) * 0.5
        frac   = max_v > 0 ? rv / max_v : 0.0
        bw     = int(bar_span * frac)
        if bw <= 0
            continue
        bx = box.new(t1, rt, t1 + bw, rb,
             bgcolor      = is_poc ? color.new(poc_css, 60) : in_va ? va_css : bar_css,
             border_color = na,
             xloc         = xloc.bar_time)
        array.push(bxs, bx)

draw_levels(float vah_p, float poc_p, float val_p,
             int t_start, string prefix, int line_width_control,

    poc_ln = line.new(t_start, poc_p, time, poc_p,
         xloc = xloc.bar_time, color = poc_css,
         width = line_width_scale + line_width_control, style = line.style_dotted, extend = extend.right)
    val_ln = line.new(t_start, val_p, time, val_p,
         xloc = xloc.bar_time, color = val_css,
         width = line_width_scale + line_width_control, style = line.style_dotted, extend = extend.right)
    array.push(lns, vah_ln)
    array.push(lns, poc_ln)
    array.push(lns, val_ln)
    if show_labels
        lx = bar_index + get_label_offset(prefix)
        sz = lbl_size()
        float y_offset = syminfo.mintick * 2
        lbl_vah = label.new(lx, vah_p + y_offset, prefix + " VAH",
             xloc = xloc.bar_index, style = label.style_label_left,
             color = color.new(vah_css, 85), textcolor = vah_css, size = sz)
        lbl_poc = label.new(lx, poc_p + y_offset, prefix + " POC",
             xloc = xloc.bar_index, style = label.style_label_left,
             color = color.new(poc_css, 85), textcolor = poc_css, size = sz)
        lbl_val = label.new(lx, val_p + y_offset, prefix + " VAL",

             color = color.new(val_css, 85), textcolor = val_css, size = sz)
