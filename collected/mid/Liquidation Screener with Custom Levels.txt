indicator('Liquidation Screener with Custom Levels 👨‍💼', overlay=true)
clback = input.int(defval=40, title='Highest/Lowest Lookback', minval=1)
custom_price = input.float(defval=1000.0, title='Custom Price Levels', minval=0)
mode = input.string(defval='VWMA Mode', options=['Long Bottom/Short Top', 'Long Top/Short Bottom', 'Top Price Liquidations', 'Bottom Price Liquidations', 'VWMA Mode', 'Custom Price'])
top_vlen = input.int(defval=90, title='Top VWMA Length', minval=1)
bottom_vlen = input.int(defval=90, title='Bottom VWMA Length', minval=1)
slen = input(defval=close, title='VWMA Source')
offsetval = input.int(defval=0, title='Offset', minval=0)
projectprice = input(defval=false, title='Track Price')
show_labels = input(defval=false, title='Show Price Labels')

mal1_ratio = input.float(defval=1.886, title='First Long MA Ratio', minval=0, step=0.01, group='Moving Averages Ratio')
mal2_ratio = input.float(defval=1.5, title='Second Long MA Ratio', minval=0, step=0.01, group='Moving Averages Ratio')
mal3_ratio = input.float(defval=1.33, title='Third Long MA Ratio', minval=0, step=0.01, group='Moving Averages Ratio')
mas1_ratio = input.float(defval=0.5, title='First Short MA Ratio', minval=0, step=0.01, group='Moving Averages Ratio')
mas2_ratio = input.float(defval=0.6, title='Second Short MA Ratio', minval=0, step=0.01, group='Moving Averages Ratio')
mas3_ratio = input.float(defval=0.67, title='Third Short MA Ratio', minval=0, step=0.01, group='Moving Averages Ratio')
gold_level = input.bool(false, 'Gold Level', group='Moving Averages Ratio')

xau = gold_level ? request.security('OANDA:XAUUSD', '1D', close)*0.00065 : 1
slen := slen/xau

swing_high = ta.highest(high, clback)
swing_low = ta.lowest(low, clback)

if mode == 'Long Bottom/Short Top'
    swing_high := ta.highest(high, clback)
    swing_low := ta.lowest(low, clback)
if mode == 'Long Top/Short Bottom'
    swing_low := ta.highest(high, clback)
    swing_high := ta.lowest(low, clback)
if mode == 'Top Price Liquidations'
    swing_low := swing_high
if mode == 'Bottom Price Liquidations'
    swing_high := swing_low
if mode == 'Custom Price'
    swing_high := custom_price
    swing_low := custom_price
if mode == 'VWMA Mode'
    swing_high := ta.vwma(slen, top_vlen)
    swing_low := ta.vwma(slen, bottom_vlen)

liq1x = swing_high * mal1_ratio
liq2x = swing_high * mal2_ratio
liq3x = swing_high * mal3_ratio

sliq1x = swing_low * mas1_ratio
sliq2x = swing_low * mas2_ratio
sliq3x = swing_low * mas3_ratio

plot(swing_high, 'Swing High', color=color.new(color.yellow, 0))

a0 = plot(liq1x, 'Short 1x', offset=offsetval, color=color.new(color.red, 0), trackprice=projectprice)
a1 = plot(liq2x, 'Short 2x', offset=offsetval, color=color.new(color.maroon, 0), trackprice=projectprice)
a2 = plot(liq3x, 'Short 3x', offset=offsetval, color=color.new(color.maroon, 0), trackprice=projectprice)

b0 = plot(sliq1x, 'Long 1x', offset=offsetval, color=color.new(color.green, 0), trackprice=projectprice)
b1 = plot(sliq2x, 'Long 2x', offset=offsetval, color=color.new(color.lime, 0), trackprice=projectprice)
b2 = plot(sliq3x, 'Long 3x', offset=offsetval, color=color.new(color.lime, 0), trackprice=projectprice)

plot(swing_low, 'Swing Low', color=color.new(color.blue, 0))

fill(a0, a1, color=color.new(color.red, 66), title='PRZ Killzone')
fill(a1, a2, color=color.new(color.maroon, 75), title='2-3x Kill Zone')
fill(b0, b1, color=color.new(color.green, 66), title='PRZ Killzone')
fill(b1, b2, color=color.new(color.lime, 75), title='2-3x Kill Zone')

if barstate.islast and show_labels
    dt = time - time[1]
    l1 = label.new(time + 10 * dt, liq1x, '💥 '+str.tostring(liq1x, '#.##'), xloc.bar_time, yloc.price, #393d45, label.style_label_left, color.orange)
    label.delete(l1[1])
    l2 = label.new(time + 10 * dt, liq2x, '🎈 '+str.tostring(liq2x, '#.##'), xloc.bar_time, yloc.price, #393d45, label.style_label_left, color.red)
    label.delete(l2[1])
    l3 = label.new(time + 10 * dt, liq3x, '🚩 '+str.tostring(liq3x, '#.##'), xloc.bar_time, yloc.price, #393d45, label.style_label_left, color.red)
    label.delete(l3[1])
    s1 = label.new(time + 10 * dt, sliq1x, '💎 '+str.tostring(sliq1x, '#.##'), xloc.bar_time, yloc.price, #393d45, label.style_label_left, color.green)
    label.delete(s1[1])
    s2 = label.new(time + 10 * dt, sliq2x, '💰 '+str.tostring(sliq2x, '#.##'), xloc.bar_time, yloc.price, #393d45, label.style_label_left, color.lime)
    label.delete(s2[1])
    s3 = label.new(time + 10 * dt, sliq3x, '💵 '+str.tostring(sliq3x, '#.##'), xloc.bar_time, yloc.price, #393d45, label.style_label_left, color.lime)
    label.delete(s3[1])
