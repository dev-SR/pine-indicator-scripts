// Core Oscillator Calculation
calcOsc(int len, int smt) =>
    float hi = ta.highest(len)
    float lo = ta.lowest(len)
    float av = ta.sma(hl2, len)
    float sig = ta.ema(ta.linreg((close - math.avg(hi, lo, av)) / (hi - lo) * 100, len, 0), smt)
    float sgD = ta.sma(sig, 2)
    [sig, sgD]

// Money Flow Calculation
calcMF(int len, int smt) =>
    float mfi = ta.sma(ta.mfi(hl2, len) - 50, smt)
    mfi

// Reversal Detection
calcReversal(float sig, float sgD, float mfi, int factor) =>
    vMA  = ta.sma(volume, 7)
    rsi  = ta.rsi(vMA, 7) - 50
    
    tMj  = volume > vMA * (factor != 10 ? 1 + (factor / 10.0) : 2.0)
    tMn  = volume > vMA * (factor != 10 ? 0 + (factor / 10.0) : 2.0) and not tMj
    
    mjBL = tMj and sig < -factor and mfi < 0
    mjBR = tMj and sig >  factor and mfi > 0
    mnBL = tMn and sig < -20 and sig < sgD and rsi < -20
    mnBR = tMn and sig >  20 and sig > sgD and rsi >  20
    
    [mjBL, mjBR, mnBL, mnBR]

// Combined Signal Calculation
calcSignals() =>
    [sig, sgD] = calcOsc(mL, sL)
    mfi = calcMF(mfL, mfS)
    [mjBL, mjBR, mnBL, mnBR] = calcReversal(sig, sgD, mfi, rsF)
    
    // Trend Direction
    trend = sig > sgD ? 1 : sig < sgD ? -1 : 0
    
    // Money Flow Direction
    mfDir = mfi > 0 ? 1 : mfi < 0 ? -1 : 0
    
    // Confluence
    conf = sig > 0 and mfi > 0 ? 1 : sig < 0 and mfi < 0 ? -1 : 0
    
    // Cross Signals
    crossUp = ta.crossover(sig, sgD)
    crossDn = ta.crossunder(sig, sgD)
    
    // Reversal Signals
    revBull = mjBL or mnBL
    revBear = mjBR or mnBR
    
    [sig, sgD, mfi, trend, mfDir, conf, crossUp, crossDn, revBull, revBear]
