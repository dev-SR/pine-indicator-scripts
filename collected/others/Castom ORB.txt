// Returns MA input selection variant, default to ALMA if blank or typo.
variant(type, src, len, offSig, offALMA) =>
    v1 = ta.alma(src, len, offALMA, offSig)  // Arnaud Legoux
    v2 = ta.ema(src, len)  // Exponential
    v3 = 2 * v2 - ta.ema(v2, len)  // Double Exponential
    v4 = 3 * (v2 - ta.ema(v2, len)) + ta.ema(ta.ema(v2, len), len)  // Triple Exponential
    v5 = ta.wma(src, len)  // Weighted
    v6 = ta.vwma(src, len)  // Volume Weighted
    v7 = 0.0
    sma_1 = ta.sma(src, len)  // Smoothed
    v7 := na(v7[1]) ? sma_1 : (v7[1] * (len - 1) + src) / len
    v8 = ta.wma(2 * ta.wma(src, len / 2) - ta.wma(src, len), math.round(math.sqrt(len)))  // Hull
    v9 = ta.linreg(src, len, offSig)  // Least Squares
    v10 = ta.sma(src, len)  // Simple
    v11 = ta.sma(v1, len)  // Triangular (extreme smooth)
    // SuperSmoother filter
    // © 2013  John F. Ehlers
    a1 = math.exp(-1.414 * 3.14159 / len)
    b1 = 2 * a1 * math.cos(1.414 * 3.14159 / len)
    c2 = b1
    c3 = -a1 * a1
    c1 = 1 - c2 - c3
    v12 = 0.0
    v12 := c1 * (src + nz(src[1])) / 2 + c2 * nz(v12[1]) + c3 * nz(v12[2])
    type == 'EMA' ? v2 : type == 'DEMA' ? v3 : type == 'TEMA' ? v4 : type == 'WMA' ? v5 : type == 'VWMA' ? v6 : type == 'SMMA' ? v7 : type == 'HullMA' ? v8 : type == 'LSMA' ? v9 : type == 'SMA' ? v10 : type == 'TMA' ? v11 : type == 'SSMA' ? v12 : v1

nphamvpa() =>
    rwhmins = (high - nz(low[2])) / (ta.atr(2) * math.sqrt(2))
    rwhmaxs = (high - nz(low[10])) / (ta.atr(10) * math.sqrt(10))
    rwhs = math.max(rwhmins, rwhmaxs)

    rwlmins = (nz(high[2]) - low) / (ta.atr(2) * math.sqrt(2))
    rwlmaxs = (nz(high[10]) - low) / (ta.atr(10) * math.sqrt(10))
    rwls = math.max(rwlmins, rwlmaxs)

    rwhminl = (high - nz(low[8])) / (ta.atr(8) * math.sqrt(8))
    rwhmaxl = (high - nz(low[40])) / (ta.atr(40) * math.sqrt(40))
    rwhl = math.max(rwhminl, rwhmaxl)

    rwlminl = (nz(high[8]) - low) / (ta.atr(8) * math.sqrt(8))
    rwlmaxl = (nz(high[40]) - low) / (ta.atr(40) * math.sqrt(40))
    rwll = math.max(rwlminl, rwlmaxl)

    //RWILLo  = max(rllmin,rllmax)

    ground = rwhs
    sky = rwls

    j = rwhl - rwll
    k = rwhs - rwls
    j2 = rwhl
    k2 = rwll

    ja = ta.crossover(j, 1)  // The following section check the diffeent condition of the RWi above and below zero
    jb = ta.crossunder(j, 1)  // In oder to check which trend is doing what
    jc = ta.crossover(-1, j)
    jd = ta.crossover(j, -1)
    j2a = ta.crossover(j2, 1)
    j2b = ta.crossunder(j2, 1)
    k2a = ta.crossover(k2, 1)
    k2b = ta.crossunder(k2, 1)
    //Define the Major, Minor and Immediate trend Status
    upmajoron = j > 1 and ja[1]
    upmajoroff = j < 1 and jb[1]

    upminoron = j2 > 1 and j2a[1]
    upminoroff = j2 < 1 and j2b[1]
    dnmajoron = j < -1 and jc[1]
    dnmajoroff = j > -1 and jd[1]
    dnminoron = k2 > 1 and k2a[1]
    dnminoroff = k2 < 1 and k2b[1]
    upmid = ground > 1 ? 1 : 0
    dnimd = sky > 1 ? 1 : 0
    iff_1 = j < -1 ? -1 : 0
    upmajor = j > 1 ? 1 : iff_1
    upminor = j2 > 1 ? 1 : -1
    dnminor = k2 > 1 ? 1 : -1

    upmajclr = upmajor == 1 ? color.rgb(3, 252, 36) : upmajor == -1 ? #fa5050 : #f4fc79
    //upmajclr = upmajor == 1 ? color.lime : upmajor == -1? color.red : color.yellow
    upmidclr = upmid == 1 ? color.lime : upmid == 0 and dnimd == 1 ? color.red : color.yellow
    upminclr = upminor == 1 ? color.lime : upminor == -1 ? color.red : color.yellow

    // Đếm số lượng màu xanh (lime) và màu đỏ
    count_lime = (upmajclr == color.lime ? 1 : 0) + (upmidclr == color.lime ? 1 : 0) + (upminclr == color.lime ? 1 : 0)
    count_red = (upmajclr == color.red ? 1 : 0) + (upmidclr == color.red ? 1 : 0) + (upminclr == color.red ? 1 : 0)

    // Kiểm tra điều kiện và đưa ra kết quả
    result = count_lime > 1 ? 1 : count_red > 1 ? 2 : 0

    result

    // // Hiển thị kết quả: Mũi tên lên khi có nhiều hơn 1 cột màu đỏ, mũi tên xuống khi có nhiều hơn 1 cột màu xanh

rsi(price, length) =>
    NetChgAvg = ta.ema(price - price[1], length)
    TotChgAvg = ta.ema(math.abs(price - price[1]), length)
    ChgRatio = TotChgAvg != 0 ? NetChgAvg / TotChgAvg : 0

    RSI = 50 * (ChgRatio + 1)

signalcal(sym) =>

    [closealt, openalt] = request.security(sym, stratRes, [closeSeries, openSeries], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)

    bool buystate = false
    bool sellstate = false  

    long = ta.crossover(closealt, openalt)
    short = ta.crossunder(closealt, openalt)

    rsi = rsi(close, rsi_length)

    //ADX Calculations 
    [diplus, diminus, adx] = ta.dmi(len, lensig)
    
    adxsig = adx >= adxtrend

    vpabuy = usevpa ? nphamvpa() == 2 : true

    vpasell = usevpa ? nphamvpa() == 1 : true

    buy = long and rsi <= 100 - rsi_strength_index and vpabuy and adxsig //and not buystate

    sell = short and rsi >= rsi_strength_index and vpasell and adxsig //and not sellstate

    signal = buy ? 1 : sell ? 0 : 2

    signal

multisignal(sym, tframe) => request.security(sym, tframe, signalcal(sym))

        for i = 1 to level_info

                if i == 1
                    tp_y := (close - atrStop) * i + close
                else
                    //tp_y := (close - atrStop) * i + close
                    tp_y := tp_y + (tp_y - close) * 0.618
            else
                tp_y := (close - atrStop) * i + close
    else
        tp_y := na

    tp_y

relmove = absmove >= threshold  // Identify "Relevant move" by comparing the absolute move to the threshold



// Bullish Order Block Identification
bullishOB = close[ob_period] < open[ob_period]  // Determine potential Bullish OB candle (red candle)

int upcandles = 0
for i = 1 to periods by 1
    upcandles += (close[i] > open[i] ? 1 : 0)  // Determine color of subsequent candles (must all be green to identify a valid Bullish OB)
    upcandles
